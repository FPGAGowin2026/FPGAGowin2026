              1.General 配置框

     General 配置框用于配置产生 IP 设计文件的相关信息。

              Device：显示已配置的芯片系列。

              Part Number：显示已配置的芯片型号。

              Module Name：配置产生的 IP 设计文件的 module name。在右侧
         文本框可重新编辑模块名称。Module Name 不能与原语名称相同，若相
         同，则报出 Error 提示。

              File Name：配置产生的 IP 设计文件的文件名。在右侧文本框可
         重新编辑文件名称。

              Create In：配置产生的 IP 设计文件的目标路径。可在右侧文本框
         中重新编辑目标路径，也可通过文本框右侧选择按钮选择目标路径。

15.2激励创建与仿真验证
     为了测试简单伪双端口 SP，可以通过实际写入一些数据再读取部分数据的
方式来验证 SP 读写是否正常。添加并新建 tb 文件命名为 ram_tb.v。编写 tb 代
码，具体 tb 代码实现的是在地址从 0~16 上写入数据为从 255 减至 240。延时一
段时间后读地址为 0~16 上的数据。

 module ram_tb();
     reg reset;
     reg clk;
     reg wea;
     reg [7:0]ad;
     reg [7:0]dina;
     wire[7:0]doutb;
     integer i;
     GSR GSR(.GSRI(1'b1));

       Gowin_RAM16SDP your_instance_name(
           .dout(doutb), //output [7:0] dout
           .clk(clk), //input clk
           .oce(), //input oce
           .ce(1), //input ce
           .reset(reset), //input reset
           .wre(wea), //input wre
           .ad(ad), //input [7:0] ad
           .din(dina) //input [7:0] din
       );
       initial clk = 1'b1;
       always #(`CLK_PERIOD/2) clk = ~clk;


 initial begin
     wea=0;
     ad=0;
     dina=0;

       #(`CLK_PERIOD*10 +1 );
       wea=1;
       for (i=0;i<=15;i=i+1)begin
           dina=255-i;
           addra = i;
       #`CLK_PERIOD;
       end
           wea=0;
           ad=0;
       #1;
       for (i=0;i<=15;i=i+1)begin
           addrb=i;
           #`CLK_PERIOD;
       end
       #200;
       $stop;
 end
