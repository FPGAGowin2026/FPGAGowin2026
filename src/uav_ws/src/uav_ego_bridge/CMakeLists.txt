cmake_minimum_required(VERSION 3.8)
project(uav_ego_bridge)

find_package(ament_cmake REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_executable(ego_sensor_bridge src/ego_sensor_bridge.cpp)
target_compile_features(ego_sensor_bridge PRIVATE cxx_std_17)
target_compile_options(ego_sensor_bridge PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(ego_sensor_bridge geometry_msgs nav_msgs rclcpp sensor_msgs
  std_msgs tf2 tf2_geometry_msgs tf2_ros)
install(TARGETS ego_sensor_bridge DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(ego_sensor_bridge_test test/test_sensor_bridge.py TIMEOUT 30)
endif()
ament_package()
