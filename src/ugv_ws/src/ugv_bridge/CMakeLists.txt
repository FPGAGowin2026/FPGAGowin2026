cmake_minimum_required(VERSION 3.8)
project(ugv_bridge)

find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(vision_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(Threads REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "srv/MissionStart.srv")
ament_export_dependencies(rosidl_default_runtime)

add_executable(ugv_bridge_node src/ugv_bridge_node.cpp)
add_executable(robot_video_node src/robot_video_node.cpp)
foreach(node ugv_bridge_node robot_video_node)
  target_compile_features(${node} PRIVATE cxx_std_17)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs)
  target_link_libraries(${node} Threads::Threads)
endforeach()
ament_target_dependencies(ugv_bridge_node tf2_ros)
ament_target_dependencies(robot_video_node vision_msgs)
target_include_directories(robot_video_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(robot_video_node ${OpenCV_LIBS})

install(TARGETS ugv_bridge_node robot_video_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/robot_command.py DESTINATION lib/${PROJECT_NAME} RENAME robot_command)
install(FILES README.md DESTINATION share/${PROJECT_NAME})

ament_package()
